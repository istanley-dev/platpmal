=== readingOrder ===
function readingOrder(){
  var keys=Object.keys(DISP);
  keys.sort(function(a,b){
    var da=a.split(' ')[0], db=b.split(' ')[0];
    var ia=(BIB_ORDER[da]!==undefined?BIB_ORDER[da]:9), ib=(BIB_ORDER[db]!==undefined?BIB_ORDER[db]:9);
    if(ia!==ib)return ia-ib;
    return (parseInt(a.split(' ')[1])||0)-(parseInt(b.split(' ')[1])||0);
  });
  return keys;
}

=== trackAnswer ===
function trackAnswer(q,a,ok,conf){
  const k=asKey(q), rec=asGet(k,q.m,q.a);
  const prev=rec.hist[rec.hist.length-1];
  // streak de acertos imediatamente anteriores
  let streak=0;for(let i=rec.hist.length-1;i>=0;i--){if(rec.hist[i].ok)streak++;else break;}
  rec.hist.push({ok:ok,ce:q.ce,nv:q.n,ts:Date.now(),round:S.round,ans:a,g:q.g,id:q.id,conf:conf||null});
  // Registro obrigatório por questão (11 campos)
  S.log.push({m:q.m,a:q.a,sub:q.a,peg:q.p||'',ans:a,gab:q.g,res:ok?'acerto':'erro',erro:ok?'':q.ce,nv:q.n,round:S.round,ts:Date.now(),status:'',conf:conf||null});
  registerQuestionStudy(q,ok,conf);
  let alert=null;
  const h=rec.hist, errRec=h.slice(-4).filter(x=>!x.ok).length;
  // FALSA CONFIANÇA: acertou mas chutando — alerta específico, prioridade alta (mesmo sem erro)
  if(ok&&conf==='chute'){
    const chutesRec=h.slice(-3).filter(x=>x.ok&&x.conf==='chute').length;
    if(chutesRec>=2){
      agenda(rec,'instavel');
      alert={tipo:'falsaconfianca',cls:'amarelo',titulo:'🎲 ACERTO NO CHUTE — ATENÇÃO',rec:rec,q:q,
             msg:'Você acertou <b>'+q.a+'</b>, mas marcou que chutou — e isso já aconteceu outras vezes nesse assunto. O placar parece bom, mas o domínio real ainda <b>não existe</b>. Esse é o tipo de lacuna que custa caro na prova real (pontuação líquida).'};
      rec.status=statusOf(rec);
      if(S.log.length)S.log[S.log.length-1].status=rec.status;
      return alert;
    }
  }
  // TÓPICO FRACO: erro recorrente no mesmo tópico-raiz, mesmo espalhado entre vários rótulos de assunto
  // (ex.: "Crase — horas do dia" + "Crase — locução" + "Crase — pronome" contam juntos como "Crase").
  if(!ok){
    const top=topico(q.a);
    const topH=topicHist(q.m,top);
    const topErrRec=topH.slice(-6).filter(x=>!x.ok).length;
    if(topErrRec>=3){
      const jaAlertado=S.topAlerted&&S.topAlerted[q.m+'▸'+top];
      const passou1Dia=!jaAlertado||(Date.now()-jaAlertado)>DIA_MS;
      if(passou1Dia){
        if(!S.topAlerted)S.topAlerted={};
        S.topAlerted[q.m+'▸'+top]=Date.now();
        alert={tipo:'topico_fraco',cls:'vermelho',titulo:'📚 TÓPICO INTEIRO EM RISCO',rec:rec,q:q,
               mat:q.m,topico:top,n:topErrRec,
               msg:'Você já errou <b>'+topErrRec+' questões de '+esc(top)+'</b> ('+q.m+') — contando as variações desse mesmo assunto, não só esta exata. Vale reforçar o tópico inteiro agora, não só esta questão.'};
        rec.status=statusOf(rec);
        if(S.log.length)S.log[S.log.length-1].status=rec.status;
        return alert;
      }
    }
  }
  if(!ok){
    if(errRec>=2){ // 1) ERRO RECORRENTE
      rec.crit=true; agenda(rec,'recorrente');
      alert={tipo:'recorrente',cls:'vermelho',titulo:'🚨 ALERTA DE ERRO RECORRENTE',rec:rec,q:q,
             msg:'Você errou novamente questões de <b>'+q.a+'</b> ('+q.m+'). Esse assunto precisa entrar em <b>revisão imediata</b>.'};
    }else if(streak>=3){ // 3) ERRO APÓS VÁRIOS ACERTOS = QUEDA (não grave)
      agenda(rec,'medio');
      const causa={'atenção':'leitura apressada / desatenção','pegadinha':'pegadinha da banca','interpretação':'interpretação do enunciado','lei seca':'esquecimento da letra da lei','jurisprudência':'jurisprudência específica','conteúdo':'lacuna pontual de conteúdo'}[q.ce]||'causa pontual';
      alert={tipo:'queda',cls:'amarelo',titulo:'📉 QUEDA DE DESEMPENHO DETECTADA',rec:rec,q:q,
             msg:'Você vinha acertando <b>'+q.a+'</b> ('+streak+' seguidas), mas errou esta. Não é fraqueza grave. Possível causa: <b>'+causa+'</b>.'};
    }else if(prev && prev.ok){ // 2) ACERTO SEGUIDO DE ERRO = INSTÁVEL
      agenda(rec,'instavel');
      const ant=[...h].reverse().find(x=>x.ok); // última acertada do mesmo assunto
      alert={tipo:'instavel',cls:'amarelo',titulo:'⚠️ CONHECIMENTO INSTÁVEL',rec:rec,q:q,antId:ant?ant.id:null,
             msg:'Você acertou uma questão de <b>'+q.a+'</b>, mas errou outra logo em seguida. O assunto ainda <b>não está consolidado</b>.'};
    }else{ agenda(rec,'leve'); }
  }else{
    // acerto: se dominado, agenda manutenção espaçada
    if(statusOf(rec)==='verde')agenda(rec,'manutencao');
  }
  rec.status=statusOf(rec);
  if(S.log.length)S.log[S.log.length-1].status=rec.status;
  return alert;
}
