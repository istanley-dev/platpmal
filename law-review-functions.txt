function computeTodaysArticles(){
  var keys=readingOrder(),key=localDateKey(new Date()),qty=dateFromKey(key).getDay()===0?5:READ_DAILY_QTY,err=lawErrorKeys(),base=scheduleForDate(key),dso=lawKeysForSubjects(base.lawFocus||base.subs||[]),used={},todays=[];
  function take(arr,n){arr.forEach(function(k){if(n>0&&!used[k]){used[k]=1;todays.push(k);n--;}});}
  function stale(a,b){var ta=S.bibLidos[a]&&S.bibLidos[a].ts||0,tb=S.bibLidos[b]&&S.bibLidos[b].ts||0;return ta-tb||lawScore(b)-lawScore(a);}
  take(keys.filter(function(k){return dso[k];}).sort(stale),3);take(keys.filter(function(k){return err[k];}).sort(stale),2);take(keys.slice().sort(stale),qty-todays.length);
  return {N:keys.length,remaining:keys.filter(function(k){return !S.bibLidos[k];}).length,perDay:todays.length,todays:todays.slice(0,qty),inReview:false};

=== readingOrder ===
function readingOrder(){
  var keys=Object.keys(DISP);
  keys.sort(function(a,b){
    var da=a.split(' ')[0], db=b.split(' ')[0];
    var ia=(BIB_ORDER[da]!==undefined?BIB_ORDER[da]:9), ib=(BIB_ORDER[db]!==undefined?BIB_ORDER[db]:9);
    if(ia!==ib)return ia-ib;
    return (parseInt(a.split(' ')[1])||0)-(parseInt(b.split(' ')[1])||0);
  });
  return keys;
}

=== trackAnswer ===
function trackAnswer(q,a,ok,conf){
  const k=asKey(q), rec=asGet(k,q.m,q.a);
  const prev=rec.hist[rec.hist.length-1];
  // streak de acertos imediatamente anteriores
  let streak=0;for(let i=rec.hist.length-1;i>=0;i--){if(rec.hist[i].ok)streak++;else break;}
  rec.hist.push({ok:ok,ce:q.ce,nv:q.n,ts:Date.now(),round:S.round,ans:a,g:q.g,id:q.id,conf:conf||null});
  // Registro obrigatório por questão (11 campos)
  S.log.push({m:q.m,a:q.a,sub:q.a,peg:q.p||'',ans:a,gab:q.g,res:ok?'acerto':'erro',erro:ok?'':q.ce,nv:q.n,round:S.round,ts:Date.now(),status:'',conf:conf||null});
  registerQuestionStudy(q,ok,conf);
  let alert=null;
  const h=rec.hist, errRec=h.slice(-4).filter(x=>!x.ok).length;
  // FALSA CONFIANÇA: acertou mas chutando — alerta específico, prioridade alta (mesmo sem erro)
  if(ok&&conf==='chute'){
    const chutesRec=h.slice(-3).filter(x=>x.ok&&x.conf==='chute').length;
    if(chutesRec>=2){
      agenda(rec,'instavel');
      alert={tipo:'falsaconfianca',cls:'amarelo',titulo:'🎲 ACERTO NO CHUTE — ATENÇÃO',rec:rec,q:q,
             msg:'Você acertou <b>'+q.a+'</b>, mas marcou que chutou — e isso já aconteceu outras vezes nesse assunto. O placar parece bom, mas o domínio real ainda <b>não existe</b>. Esse é o tipo de lacuna que custa caro na prova real (pontuação líquida).'};
      rec.status=statusOf(rec);
      if(S.log.length)S.log[S.log.length-1].status=rec.status;
      return alert;
    }
  }
  // TÓPICO FRACO: erro recorrente no mesmo tópico-raiz, mesmo espalhado entre vários rótulos de assunto
  // (ex.: "Crase — horas do dia" + "Crase — locução" + "Crase — pronome" contam juntos como "Crase").
  if(!ok){
    const top=topico(q.a);
    const topH=topicHist(q.m,top);
    const topErrRec=topH.slice(-6).filter(x=>!x.ok).length;
    if(topErrRec>=3){
      const jaAlertado=S.topAlerted&&S.topAlerted[q.m+'▸'+top];
      const passou1Dia=!jaAlertado||(Date.now()-jaAlertado)>DIA_MS;
      if(passou1Dia){
        if(!S.topAlerted)S.topAlerted={};
        S.topAlerted[q.m+'▸'+top]=Date.now();
        alert={tipo:'topico_fraco',cls:'vermelho',titulo:'📚 TÓPICO INTEIRO EM RISCO',rec:rec,q:q,
               mat:q.m,topico:top,n:topErrRec,
               msg:'Você já errou <b>'+topErrRec+' questões de '+esc(top)+'</b> ('+q.m+') — contando as variações desse mesmo assunto, não só esta exata. Vale reforçar o tópico inteiro agora, não só esta questão.'};
        rec.status=statusOf(rec);
        if(S.log.length)S.log[S.log.length-1].status=rec.status;
        return alert;
      }
    }
  }
  if(!ok){
    if(errRec>=2){ // 1) ERRO RECORRENTE
      rec.crit=true; agenda(rec,'recorrente');
      alert={tipo:'recorrente',cls:'vermelho',titulo:'🚨 ALERTA DE ERRO RECORRENTE',rec:rec,q:q,
             msg:'Você errou novamente questões de <b>'+q.a+'</b> ('+q.m+'). Esse assunto precisa entrar em <b>revisão imediata</b>.'};
    }else if(streak>=3){ // 3) ERRO APÓS VÁRIOS ACERTOS = QUEDA (não grave)
      agenda(rec,'medio');
      const causa={'atenção':'leitura apressada / desatenção','pegadinha':'pegadinha da banca','interpretação':'interpretação do enunciado','lei seca':'esquecimento da letra da lei','jurisprudência':'jurisprudência específica','conteúdo':'lacuna pontual de conteúdo'}[q.ce]||'causa pontual';
      alert={tipo:'queda',cls:'amarelo',titulo:'📉 QUEDA DE DESEMPENHO DETECTADA',rec:rec,q:q,
             msg:'Você vinha acertando <b>'+q.a+'</b> ('+streak+' seguidas), mas errou esta. Não é fraqueza grave. Possível causa: <b>'+causa+'</b>.'};
    }else if(prev && prev.ok){ // 2) ACERTO SEGUIDO DE ERRO = INSTÁVEL
      agenda(rec,'instavel');
      const ant=[...h].reverse().find(x=>x.ok); // última acertada do mesmo assunto
      alert={tipo:'instavel',cls:'amarelo',titulo:'⚠️ CONHECIMENTO INSTÁVEL',rec:rec,q:q,antId:ant?ant.id:null,
             msg:'Você acertou uma questão de <b>'+q.a+'</b>, mas errou outra logo em seguida. O assunto ainda <b>não está consolidado</b>.'};
    }else{ agenda(rec,'leve'); }
  }else{
    // acerto: se dominado, agenda manutenção espaçada
    if(statusOf(rec)==='verde')agenda(rec,'manutencao');
  }
  rec.status=statusOf(rec);
  if(S.log.length)S.log[S.log.length-1].status=rec.status;
  return alert;
}

function getSmartPool(pool){
  // 1) não vistas > 2) erradas recentes > 3) erradas antigas > 4) acertos antigos > 5) tudo
  const vistas=new Set(S.ha.map(a=>a.id));
  const erradas=new Set([...S.eids]);
  const recentes=new Set(S.ha.slice(-50).map(a=>a.id)); // últimas 50 respondidas = "recentes"
  const naoVistas=pool.filter(q=>!vistas.has(q.id));
  if(naoVistas.length>=3)return naoVistas; // prioridade 1
  const erradasP=pool.filter(q=>erradas.has(q.id)&&!recentes.has(q.id));
  if(erradasP.length>=3)return erradasP; // prioridade 2
  const erradasRecentes=pool.filter(q=>erradas.has(q.id));
  if(erradasRecentes.length>=3)return erradasRecentes; // prioridade 3
  const semRecentes=pool.filter(q=>!recentes.has(q.id));
  if(semRecentes.length>=3)return semRecentes; // evita repetir recentes
  return pool; // tudo disponível
}
function getPool(){
  let p=QQ.filter(function(q){return questionUsable(q)&&!q.reviewOnly&&q.scope!=='review_only';});
  if(S.mode==='erros')p=QQ.filter(q=>S.eids.has(q.id));
  else if(S.mode==='acertos'){const ok=new Set(S.ha.filter(a=>a.ok).map(a=>a.id));p=QQ.filter(q=>ok.has(q.id));}
  else if(S.mode==='oficiais')p=QQ.filter(isOfficialQuestion);
  else if(S.mode==='recentes')p=QQ.filter(isRecentAdapted);
  else if(S.mode==='materia'&&S.subs.length>0)p=QQ.filter(q=>S.subs.includes(q.m));
  // Filtragem por faixas CEBRASPE (inclusive)
  if(S.niv!=='all'){
    if(S.niv==='dificil')p=p.filter(q=>['dificil','muito_dificil','cruel'].includes(q.n));
    else if(S.niv==='muito_dificil')p=p.filter(q=>['muito_dificil','cruel'].includes(q.n));
    else if(S.niv==='cruel'){
      const crueis=p.filter(q=>q.n==='cruel');
      p=crueis.length>=3?crueis:p.filter(q=>q.n==='cruel'||q.n==='muito_dificil');
    }
    else p=p.filter(q=>q.n===S.niv);
  }
  return p;
}

const BACKUP_APP='PMAL-CEBRASPE';

const BACKUP_SECRET_KEYS=new Set(['pmal26_key']);
function collectProgressStorage(){
  const data={};
  for(let i=0;i<localStorage.length;i++){
    const key=localStorage.key(i);
    if(key&&key.indexOf('pmal26')===0&&!BACKUP_SECRET_KEYS.has(key))data[key]=localStorage.getItem(key);
  }
  if(!Object.prototype.hasOwnProperty.call(data,'pmal26')){
    data.pmal26=JSON.stringify({ha:S.ha,eids:[...S.eids],AS:S.AS,round:S.round,log:S.log.slice(-500),metas:S.metas,xp:S.xp,streak:S.streak,lastDay:S.lastDay,plan:S.plan,daily:S.daily,bibLidos:S.bibLidos,leituraDia:S.leituraDia,flash:S.flash,weekly:S.weekly,reviewSessions:S.reviewSessions||{},studyHistory:S.studyHistory||{},adaptiveReview:S.adaptiveReview||{completed:{},deferred:{}},lawMemory:S.lawMemory||{cards:{},daily:{}},legacyEids:[...S.legacyEids],legacyTopics:S.legacyTopics,legacySummary:S.legacySummary,cycleId:S.cycleId});
  }
  return data;
}
function downloadProgressDump(dump,suffix){
  const blob=new Blob([JSON.stringify(dump,null,2)],{type:'application/json'});
  const url=URL.createObjectURL(blob),a=document.createElement('a'),d=new Date();
  const stamp=d.getFullYear()+'-'+String(d.getMonth()+1).padStart(2,'0')+'-'+String(d.getDate()).padStart(2,'0')+'-'+String(d.getHours()).padStart(2,'0')+String(d.getMinutes()).padStart(2,'0');
  a.href=url;a.download='pmal-backup-'+stamp+(suffix||'')+'.json';
  document.body.appendChild(a);a.click();
  setTimeout(()=>{URL.revokeObjectURL(url);a.remove();},500);
}
function exportProgress(suffix){
  try{
    const dump={_app:BACKUP_APP,_v:BACKUP_VERSION,_exportedAt:new Date().toISOString(),storage:collectProgressStorage()};
    downloadProgressDump(dump,suffix);
    toast('💾 Backup completo exportado! Guarde esse arquivo.');
    return true;
  }catch(e){toast('Não foi possível exportar: '+(e.message||e));return false;}
}
function normalizeProgressDump(dump){
  if(!dump||dump._app!==BACKUP_APP)throw new Error('Arquivo não reconhecido pela plataforma.');
  let data=dump.storage;
  if(!data&&dump.pmal26){
    data={pmal26:dump.pmal26};
    ['pmal26_gen','pmal26_red_txt','pmal26_red_tema'].forEach(function(k){if(Object.prototype.hasOwnProperty.call(dump,k))data[k]=dump[k];});
  }
  if(!data||Object.prototype.toString.call(data)!=='[object Object]'||typeof data.pmal26!=='string')throw new Error('Backup sem dados de progresso.');
  const keys=Object.keys(data);
  if(!keys.length||keys.some(function(k){return k.indexOf('pmal26')!==0||BACKUP_SECRET_KEYS.has(k)||typeof data[k]!=='string';}))throw new Error('Backup contém chaves inválidas.');
  const main=JSON.parse(data.pmal26);
  if(!main||Object.prototype.toString.call(main)!=='[object Object]')throw new Error('Progresso principal inválido.');
  if(data.pmal26_gen)JSON.parse(data.pmal26_gen);
  return data;
}
function importProgressFile(input){
  const file=input.files&&input.files[0];
  if(!file)return;
  if(file.size>10*1024*1024){toast('❌ O arquivo excede o limite de 10 MB.');input.value='';return;}
  const reader=new FileReader();
  reader.onload=function(ev){
    try{
      const dump=JSON.parse(ev.target.result);
      const data=normalizeProgressDump(dump);
      if(!confirm('Importar este backup validado? As chaves presentes nele substituirão as equivalentes neste navegador. Nenhum outro dado será apagado e uma cópia do progresso atual será baixada antes.')){input.value='';return;}
      if(!exportProgress('-antes-da-importacao'))throw new Error('A cópia de segurança atual não pôde ser criada. A importação foi cancelada.');
      const previous={};
      Object.keys(data).forEach(function(k){previous[k]=localStorage.getItem(k);});
      try{Object.keys(data).forEach(function(k){localStorage.setItem(k,data[k]);});}
      catch(writeError){
        Object.keys(previous).forEach(function(k){if(previous[k]===null)localStorage.removeItem(k);else localStorage.setItem(k,previous[k]);});
        throw new Error('Não houve espaço para concluir a importação; o progresso anterior foi restaurado.');
      }
      input.value='';
      toast('✅ Backup importado com segurança! Recarregando…');
      setTimeout(()=>location.reload(),900);
    }catch(e){toast('❌ Erro ao ler o arquivo: '+(e.message||e));input.value='';}
  };
  reader.onerror=function(){toast('❌ Não foi possível ler o arquivo.');input.value='';};
  reader.readAsText(file);
}

    // o estado ativo for reduzido, liberando espaço — esse era o ponto que travava o botão.
    var archiveMode='pendente';
    try{localStorage.setItem(archiveKey,raw);archiveMode='completo';}catch(quotaErr){console.warn('Storage cheio; arquivo será tentado após o reset:',quotaErr);}

    var sig=collectLegacySignals(snap);
    if(!(S.legacyEids instanceof Set))S.legacyEids=new Set(Array.isArray(S.legacyEids)?S.legacyEids:[]);
    if(!S.legacyTopics||typeof S.legacyTopics!=='object')S.legacyTopics={};
    Array.from(sig.eids).forEach(function(id){S.legacyEids.add(id);});
    Object.keys(sig.topics).forEach(function(k){S.legacyTopics[k]=(S.legacyTopics[k]||0)+sig.topics[k];});
    S.legacySummary=summary;

    // Zera somente o que pertence às métricas do ciclo. Preferências, data da prova,
    // redações e itens sinalizados continuam preservados.
    S.ha=[];S.eids=new Set();S.AS={};S.log=[];S.round=0;S.metas={};S.xp=0;S.streak=0;S.lastDay='';
    S.daily={d:'',q:0,rev:false,sim:false,red:false,metas:0,bonus:false,diaAnterior:false,leisFeitas:false};
    S.bibLidos={};S.leituraDia={d:'',keys:[],inReview:false};S.flash={cards:{}};S.weekly={lastDone:0};
    S.coverageQueue=[];S.weeklyAdjust=[];
    S.cycleId='ciclo_'+Date.now();
    var newCycleId=S.cycleId;

    SS();
    var saved=JSON.parse(localStorage.getItem('pmal26')||'{}');
    if(saved.cycleId!==newCycleId)throw new Error('O novo estado não foi persistido no navegador.');

    // Se a primeira cópia falhou por cota, o reset acima tornou pmal26 muito menor.
    if(archiveMode==='pendente')archiveMode=saveFullOrCompactCycleArchive(archiveKey,raw,summary);
    summary.archiveMode=archiveMode;
    appendCycleHistory(summary);

    toast('✅ Novo ciclo iniciado · histórico '+(archiveMode==='completo'?'arquivado':'preservado em modo '+archiveMode)+'.');
    setTimeout(function(){location.reload();},450);
  }catch(e){
    console.error('Falha ao iniciar novo ciclo:',e);
    toast('❌ Não foi possível iniciar novo ciclo: '+(e&&e.message?e.message:'erro de armazenamento'));
  }
}
