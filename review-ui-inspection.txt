function adaptiveReviewQuestions(item){
  var ids=[];(item.sourceDates||[item.sourceDate]).filter(Boolean).forEach(function(d){var session=S.reviewSessions&&S.reviewSessions[d];if(session&&Array.isArray(session.questionIds))ids=mergeUnique(ids,session.questionIds);});if(item.study&&item.study.reviewQuestionIds)ids=mergeUnique(ids,item.study.reviewQuestionIds);
  if(item.kind==='combat'){Object.values(S.AS||{}).filter(function(r){return r.m===item.subject&&topico(r.a)===item.topic;}).forEach(function(r){(r.hist||[]).forEach(function(h){if(!h.ok&&h.id)ids=mergeUnique(ids,[h.id]);});});}
  var qs=ids.map(findQuestionById).filter(function(q){return q&&q.m===item.subject&&!q.missingContext;});
  if(item.kind!=='d1'&&item.kind!=='dso'){
    var session=S.reviewSessions&&S.reviewSessions[item.sourceDate];
    var attempts=session&&session.attempts||[];
    function score(q){var own=attempts.filter(function(a){return a.id===q.id;}),last=own[own.length-1];if(last&&!last.ok)return 0;if(last&&(last.conf==='duvida'||last.conf==='chute'))return 1;if(!last&&(q.reviewClass==='erro'||q.reviewClass==='inseguro'))return 2;if(!last)return 3;return 4;}
    qs.sort(function(a,b){return score(a)-score(b);});
    var urgent=qs.filter(function(q){return score(q)<4;}),stable=qs.filter(function(q){return score(q)>=4;}),maintenanceSlots=urgent.length?Math.max(2,MAX_ADAPTIVE_QUESTIONS-Math.min(10,urgent.length)):2;
    qs=urgent.slice(0,10).concat(stable.slice(0,maintenanceSlots));
  }
  return qs.slice(0,MAX_ADAPTIVE_QUESTIONS);
}

function buildOfficialDailyReview(N){
  var pool=QQ.filter(function(q){return isOfficialQuestion(q)&&!isFlagged(q);});
  if(!pool.length)return [];
  var used=new Set(),nErro=Math.round(N*.50),nBranco=Math.round(N*.25),nInst=Math.round(N*.15),nMan=N-nErro-nBranco-nInst,sq=[];
  sq=sq.concat(takeOfficialBucket(pool.filter(function(q){return officialReviewClass(q)==='erro';}),nErro,used));
  sq=sq.concat(takeOfficialBucket(pool.filter(function(q){return officialReviewClass(q)==='branco';}),nBranco,used));
  sq=sq.concat(takeOfficialBucket(pool.filter(function(q){return officialReviewClass(q)==='instavel';}),nInst,used));
  sq=sq.concat(takeOfficialBucket(pool.filter(function(q){return officialReviewClass(q)==='manutencao';}),nMan,used));
  if(sq.length<N)sq=sq.concat(takeOfficialBucket(pool,N-sq.length,used));
  sq=diversifyMission(sq,N);
  return sq.slice(0,N).sort(function(){return Math.random()-.5;});
}

function buildYesterdayReviewDeck(){
  const y=new Date();y.setDate(y.getDate()-1);y.setHours(0,0,0,0);
  const yStart=y.getTime();const yEnd=yStart+DIA_MS-1;
  const seen={};
  // Fonte principal: histórico real (inclui teoria e estudos feitos fora da plataforma).
  // Matemática aparece na camada adaptativa como teoria e nunca entra neste baralho de questões.
  studySubjects(localDateKey(y)).forEach(function(rec){
    if(rec.subject==='Matemática')return;
    var tops=(rec.topics||[]).length?rec.topics:['Revisão geral'];
    tops.forEach(function(t){var k=rec.subject+'▸'+t;seen[k]={mat:rec.subject,topico:t,teveErro:(rec.errors||0)>0||(rec.insecure||0)>0};});
  });
  // Compatibilidade com o histórico antigo, anterior ao campo studyHistory.
  (S.log||[]).forEach(function(entry){
    if(entry.ts>=yStart&&entry.ts<=yEnd&&entry.m!=='Matemática'){
      const top=topico(entry.a),k=entry.m+'▸'+top;
      if(!seen[k])seen[k]={mat:entry.m,topico:top,teveErro:false};
      if(entry.res==='erro')seen[k].teveErro=true;
    }
  });
  let entries=Object.values(seen);
  entries.sort(function(a,b){return (b.teveErro?1:0)-(a.teveErro?1:0);});
  if(entries.length>MAX_ONTEM_CARDS)entries=entries.slice(0,MAX_ONTEM_CARDS);
  const cards=[];
  entries.forEach(function(e){
    const lesson=getLessonContent(e.mat,e.topico);
    if(!lesson)return;
    const fk=flashKey(e.mat,e.topico),frec=flashGet(fk);
    cards.push(Object.assign(lesson,{status:'ontem',fkey:fk,due:frec.due}));
  });
  return cards;
}
