function buildKeyToQids(){

  KEY_TO_QIDS={};

function readingOrder(){
  var keys=Object.keys(DISP);
  keys.sort(function(a,b){
    var da=a.split(' ')[0], db=b.split(' ')[0];
    var ia=(BIB_ORDER[da]!==undefined?BIB_ORDER[da]:9), ib=(BIB_ORDER[db]!==undefined?BIB_ORDER[db]:9);
    if(ia!==ib)return ia-ib;
    return (parseInt(a.split(' ')[1])||0)-(parseInt(b.split(' ')[1])||0);
  });
  return keys;
}

function startLeituraQuiz(){
  if(pendingWeekReview()){startWeekReviewGated('startLeituraQuiz');return;}
  if(pendingYesterdayReview()){startYesterdayReview('startLeituraQuiz');return;}
  if(pendingLawReview()){startLawReviewGated('startLeituraQuiz');return;}

  if(!todaysKeys.length){toast('Nenhum artigo novo hoje — você está em dia ou na fase de revisão. Treine pela Biblioteca Legal!');return;}

  var idSet={};
